# Crontab of the stack user (install with: crontab scripts/crontab.txt, or add lines with crontab -e).
# The VM runs on UTC. Scripts live in /docker/servarr/scripts/; logs go to the user's home folder.

# Sonarr: search missing episodes in small batches, max 2 searches at a time.
*/10 * * * * /usr/bin/python3 /docker/servarr/scripts/sonarr_throttled_search.py >> $HOME/sonarr-throttled-search.log 2>&1

# Sonarr: stall watchdog (cancel queued searches, then restart Sonarr; Slack alerts via /docker/servarr/slack.env).
*/5 * * * * /usr/bin/python3 /docker/servarr/scripts/sonarr_watchdog.py >> $HOME/sonarr-watchdog.log 2>&1

# Tdarr: final HEVC load test. Waits until conversion is finished, then removes its own line.
15 12 * * * /usr/bin/python3 /docker/servarr/scripts/hevc_loadtest.py --when-done >> $HOME/hevc-loadtest/cron.log 2>&1

# Tdarr worker window. Backlog mode: 1 worker 06:30-22:00 Eastern, 4 workers overnight.
# After the queue drains: normal mode, schedule 01:00-07:00 and 0 workers 06:30-07:00 (window ends 06:30).
*/5 * * * * /usr/bin/python3 /docker/servarr/scripts/tdarr_worker_window.py

# Slack reports (scripts/slack/slack_reports.py). The VM runs on UTC and Debian cron has no CRON_TZ,
# so each daily/weekly job starts at both possible UTC hours and --local-hour keeps the one that is
# 08:00 / 07:00 / 18:00 Eastern. Daylight saving is handled without editing these lines.
*/5 * * * * /usr/bin/python3 /docker/servarr/scripts/slack_reports.py health >> $HOME/slack-reports.log 2>&1
0 12,13 * * * /usr/bin/python3 /docker/servarr/scripts/slack_reports.py digest --local-hour 8 >> $HOME/slack-reports.log 2>&1
0 22,23 * * 0 /usr/bin/python3 /docker/servarr/scripts/slack_reports.py weekly --local-hour 18 >> $HOME/slack-reports.log 2>&1
# Usage stats: weekly, Friday 07:00 Eastern (covers the previous Friday to Thursday).
0 11,12 * * 5 /usr/bin/python3 /docker/servarr/scripts/slack_reports.py usage --local-hour 7 >> $HOME/slack-reports.log 2>&1
